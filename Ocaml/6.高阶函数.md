# 高阶函数



其实就是函数可以作为值来传递



比如

let square x = x*x;;

let twice f x = f f x;

let double_square x = twice square x;;

or even 

let double_square = twice square;; 部分应用



## map函数

let rec map f = function

|[]->[]

| h::t -> f h::map f t

example

map f [a;b;c]

=> [f(a);f(b);f(c)]

## Fold函数

let rec fold init op = function

| []-> init

| h::t -> op(h,fold(init,op,t))

example:

fold init f [a;b;c]=>f(a,f(b,f(c,init)))
