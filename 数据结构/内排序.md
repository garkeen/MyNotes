Shell排序

Shell 排序的复杂度取决于增量序列。同一个数组，换一组增量，最坏复杂度可能完全不同。

```C
void shell_sort(int a[], int n) {
    /* ===== 增量序列的生成 ===== */
    (...) {
        for (int i = gap; i < n; ++i) {
            int key = a[i];
            int j = i - gap;

            while (j >= 0 && a[j] > key) {
                a[j + gap] = a[j];
                j -= gap;
            }
            a[j + gap] = key;

    }
}
```

```C
void selection_insertion_sort(int a[], int n) {
    for (int i = 0; i < n - 1; ++i) {
        int min_pos = i;
        for (int j = i + 1; j < n; ++j) {
            if (a[j] < a[min_pos]) {
                min_pos = j;
```

堆排序

堆排序使用堆排序，看起来是工作在树上，但由于是完全二叉树，可以用数组存储，所以全程只在数组上操作。

```C
void sift_down(int a[], int i, int n) {
    while (1) {
        int largest = i;
        int l = 2 * i + 1;
        int r = 2 * i + 2;

        if (l < n && a[l] > a[largest]) largest = l;
        if (r < n && a[r] > a[largest]) largest = r;

        if (largest == i) break;

        int t = a[i];
        a[i] = a[largest];
        a[largest] = t;

        i = largest;
    }
}
void sift_up(int a[], int i) {
    int key = a[i];

    while (i > 0 && a[(i - 1) / 2] < key) {
        a[i] = a[(i - 1) / 2];
        i = (i - 1) / 2;
    }
    a[i] = key;
}
```

#### 快速排序

Hoare划分

```C
void quicksort(int a[], int l, int r)
{
    if (l >= r) return;                      // 0 或 1 个元素，已有序

    int i = l, j = r, p = a[l];              // 取最左元素为基准
    while (i < j) {
        while (i < j && a[j] >= p) --j;      // 右端找一个 < 基准的
        while (i < j && a[i] <= p) ++i;      // 左端找一个 > 基准的
        if (i < j) { 
            int t = a[i]; a[i] = a[j]; a[j] = t; 
        }  									// 交换a[i]和a[j]后，满足a[l->i]<=p a[j->r]>=p
    }
    a[l] = a[i]; 
    a[i] = p;                  				 // 跳出来一定有i==j,因为j优先确定，所以有a[i]<=a[l] 交换即可

    quicksort(a, l, i - 1);                  // 递归左半部分
    quicksort(a, i + 1, r);                  // 递归右半部分
}
```

##### 二路归并

自顶向下的实现

```C
void binary_merge(int a[], int t[], int lo, int mid, int hi) {
    int i = lo, j = mid + 1, k = lo;
    while (i <= mid && j <= hi) {
        if (a[i] <= a[j]) t[k] = a[i], i++;
        else              t[k] = a[j], j++;
        k++;
    }
    while (i <= mid) t[k] = a[i], i++, k++;
    while (j <= hi)  t[k] = a[j], j++, k++;
    for (k = lo; k <= hi; k++) a[k] = t[k];
}

void merge_sort(int a[], int t[], int lo, int hi) {
    if (lo >= hi) return;
    int mid = (lo + hi) / 2;
    merge_sort(a, t, lo, mid),；
    merge_sort(a, t, mid + 1, hi);
    binary_merge(a, t, lo, mid, hi);
}
```

自底向上的实现

```C
void merge_sort(int a[], int temp[], int lo, int hi) {
    int n = hi - lo + 1;
    if (n <= 1) return;

    for (int width = 1; width < n; width *= 2) {
        for (int start = lo; start <= hi - width; start += 2 * width) {
            int mid = start + width - 1;
            int end = start + 2 * width - 1;
            if (end > hi) end = hi;
            binary_merge(a, temp, start, mid, end);
        }
    }
}
```

##### K路归并

假设K个子段已经有序，那么该如何合并？缓冲区依然和原数组同样大小，需要维护k个指针，每次只取k个中最小的，然后移动对应指针，直到所有指针达到尾部。

```C
/* 缓冲区大小要求：
 *   tmp    : n			val    : 2 * Kmax			tree   : 2 * Kmax
 *   pos    : Kmax		segPtr : Kmax				segLen : Kmax
 */
void kWayMergeSort(int *a, int n, int Kmax,
                   int *tmp, int *val, int *tree, int *pos,
                   int **segPtr, int *segLen)
{
    if (!a || n <= 1) return;
    int K = (Kmax < 2) ? 2 : ((Kmax < n) ? Kmax : n);

    for (long long width = 1; width < n; width *= K) {
        for (long long base = 0; base < n; base += (long long)K * width) {
            int M = 0;
            long long total = 0;
            for (int i = 0; i < K; i++) {
                long long s = base + (long long)i * width;
                if (s >= n) break;
                long long e = s + width;
                if (e > n) e = n;
                segPtr[M] = a + s;
                segLen[M] = (int)(e - s);
                total += segLen[M];
                M++;
            }
            if (M <= 1) continue;

            kWayMerge(tmp, segPtr, segLen, M, val, tree, pos);
            memcpy(a + base, tmp, (size_t)total * sizeof(int));
        }
    }
}
```

如果使用最朴素的方法，每次都遍历全部k个元素选择最小，那么单次的复杂度是$O(k)$，一趟的复杂度是$O(nk)$，一共跑$\log_k n$趟，总共的时间复杂度是$O(nk\log_k n)$，当k越来越大时，常数项会越来越大。

```C
void kWayMerge(int *out, int **arr, int *len, int K,
               int *val, int *tree, int *pos)
{
    (void)val; (void)tree;          /* 扫描实现不需要败者树 */

    int total = 0;
    for (int i = 0; i < K; i++) { pos[i] = 0; total += len[i]; }

    for (int idx = 0; idx < total; idx++) {
        int best = -1;
        for (int i = 0; i < K; i++) {
            if (pos[i] < len[i]) {
                if (best < 0 || arr[i][pos[i]] < arr[best][pos[best]])
                    best = i;
            }
        }
        out[idx] = arr[best][pos[best]];
        pos[best]++;
    }
}
```

如果使用堆/败者树动态维护，那么单次的复杂度是$O(\log k)$,一趟的复杂度是$O(n\log k)$，一共跑$\log_k n$趟，总共的时间复杂度是$O(n\log n)$，时间复杂度与k无关。

```C
/* 调用者提供：val[2K]、tree[2K]、pos[K]。哨兵用 INT_MAX。 */
void kway_merge(int *out, int **arr, int *len, int K,
               int *val, int *tree, int *pos)
{
    const int INF = INT_MAX;
    int total = 0;
    for (int i = 0; i < K; i++) {
        pos[i] = 0;
        val[K + i] = (len[i] > 0) ? arr[i][0] : INF;
        total += len[i];
    }
    loser_init(K, val, tree);

    for (int idx = 0; idx < total; idx++) {
        int w = tree[0];                 /* 胜者叶子下标 */
        int i = w - K;                   /* 对应第几路 */
        out[idx] = val[w];
        pos[i]++;
        int nv = (pos[i] < len[i]) ? arr[i][pos[i]] : INF;
        loser_update_champion(val, tree, w, nv);
    }
}
```

```C
void radix_sort_lsd(int n, int a[], int BASE) {
    if (n <= 1 || BASE < 2) return;

    int BITS = sizeof(int) * 8;
    unsigned SIGN = 1u << (BITS - 1);

    unsigned long long limit = 1;
    for (int i = 0; i < BITS; ++i) limit *= 2;

    int R = 0;
    unsigned long long p = 1;
    while (p < limit) { p = p > limit / BASE ? limit : p * BASE; ++R; }

    int *arena = malloc(((size_t)n + BASE) * sizeof(int));
    int *b = arena, *cnt = arena + n;
    unsigned long long pw = 1;

    for (int r = 0; r < R; ++r) {
        int *src = (r & 1) ? b : a, *dst = (r & 1) ? a : b;
        for (int j = 0; j < BASE; ++j) cnt[j] = 0;
        for (int j = 0; j < n; ++j)
            ++cnt[(int)(((unsigned)src[j] ^ SIGN) / pw % BASE)];
        for (int sum = 0, j = 0; j < BASE; ++j) {
            sum += cnt[j];
            cnt[j] = sum - cnt[j];
        }
        for (int j = 0; j < n; ++j)
            dst[cnt[(int)(((unsigned)src[j] ^ SIGN) / pw % BASE)]++] = src[j];
        pw *= BASE;
    }

    if (R & 1)
        for (int j = 0; j < n; ++j) a[j] = b[j];

    free(arena);
}
```

最高位优先MSD

这里就比较难了，因为对最高位排序后，对次高位排序不能全部一起排，会打乱，必须相同的最高位内部排序。

这就相当于第一次我们分成了最多N个桶，第二次在N个桶内分别排序，继续分成了最多N*N个桶，反复如此。

所以我们需要存储每个桶的头和尾，所以要使用递归（使用等价的栈实现也可以，但是递归更好写），你当然还可以扫描前缀来识别当前处理的桶的首和尾，但是效率更低。因此MSD基数排序倾向于递归实现。

```C
static void msd(int *src, int *dst, int *bucket, int lo, int hi, int BASE,
                unsigned long long pw, unsigned SIGN) {
    int n = hi - lo;
    if (n <= 1 || !pw) return;

    /* 计数 → 前缀和 = 桶起点（相对 lo） */
    for (int j = 0; j < BASE; ++j) bucket[j] = 0;
    for (int i = lo; i < hi; ++i)
        ++bucket[(int)(((unsigned)src[i] ^ SIGN) / pw % BASE)];
    for (int j = 1; j < BASE; ++j) bucket[j] += bucket[j - 1];

    /* 从后往前分散，bucket[d] 边放边减，放完退回桶起点 */
    for (int i = hi - 1; i >= lo; --i) {
        int d = (int)(((unsigned)src[i] ^ SIGN) / pw % BASE);
        dst[lo + --bucket[d]] = src[i];
    }

    /* 此时 bucket 即桶边界：桶 j 区间 [lo+bucket[j], lo+bucket[j+1]) */
    for (int j = 0; j < BASE; ++j) {
        int l = lo + bucket[j];
        int r = lo + (j + 1 < BASE ? bucket[j + 1] : n);
        if (r > l) msd(dst, src, bucket + BASE, l, r, BASE, pw / BASE, SIGN);
    }

    for (int i = lo; i < hi; ++i) src[i] = dst[i];
}

void radix_sort_msd(int n, int a[], int BASE) {
    if (n <= 1 || BASE < 2) return;

    int BITS = sizeof(int) * 8;
    unsigned SIGN = 1u << (BITS - 1);

    unsigned long long limit = 1;
    for (int i = 0; i < BITS; ++i) limit *= 2;

    int R = 0;
    unsigned long long p = 1;
    while (p < limit) { p = p > limit / BASE ? limit : p * BASE; ++R; }

    /* 一次性申请：tmp[n] + bucket[R * BASE] */
    int *arena = malloc(((size_t)n + (size_t)R * BASE) * sizeof(int));
    if (!arena) return;
    int *tmp = arena;
    int *bucket = arena + n;

    unsigned long long pw = 1;
    while (pw < limit) pw *= BASE;
    pw /= BASE;

    msd(a, tmp, bucket, 0, n, BASE, pw, SIGN);
    free(arena);
}
```

#### 计数排序

假设所有值都必须出现在一个已知的集合内，那么我们就可以使用这样的值域数组进行计数

所以我们需要两个数组 一个是待排序数组 一个是已经按从小到大排列的值域数组

此外，计数排序需要缓冲区用来维护稳定性的输出

如果值域不是连续的，就需要二分查找，如果是连续的，直接返回就行，这里我们用index_of抽象

```C
static int index_of(const int *d, int m, int v) {
    int lo = 0, hi = m - 1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (d[mid] == v) return mid;
        if (d[mid] < v) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}
```

```C
void counting_sort(int *arr, int n, const int *domain, int m) {
    int i, k;
    int *cnt, *tmp;
    if (!arr || !domain || n <= 1 || m <= 0) return;

    cnt = calloc(m, sizeof *cnt);
    tmp = malloc(n * sizeof *tmp);

    for (i = 0; i < n; i = i + 1) {
        k = index_of(domain, m, arr[i]);
        cnt[k] = cnt[k] + 1;
    }
    for (i = 1; i < m; i = i + 1)
        cnt[i] = cnt[i] + cnt[i - 1];

    for (i = n - 1; i >= 0; i = i - 1) {
        k = index_of(domain, m, arr[i]);
        cnt[k] = cnt[k] - 1;
        tmp[cnt[k]] = arr[i];
    }
    for (i = 0; i < n; i = i + 1)
        arr[i] = tmp[i];

    free(cnt);
    free(tmp);
}
```

```C
    int i, k, running, c;
```

#### 桶排序

桶排序严格意义上不是单独的算法，而是思想，比如用于MSD基数排序，需要维护桶。

桶排序的关键步骤是把数据放到一个个桶，然后桶内部排序就行了，桶之间的位置不再改变。

这意味着我们还需要决定如何分桶，桶内部使用什么排序算法。
